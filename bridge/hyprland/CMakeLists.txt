cmake_minimum_required(VERSION 3.25)
project(writy_hyprland_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(GNUInstallDirs)
include(CTest)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HYPRLAND REQUIRED IMPORTED_TARGET hyprland)
add_library(writy-hyprland MODULE plugin.cpp)
set_target_properties(writy-hyprland PROPERTIES PREFIX "")
target_link_libraries(writy-hyprland PRIVATE PkgConfig::HYPRLAND)
target_compile_options(writy-hyprland PRIVATE -Wall -Wextra -Wpedantic -fno-gnu-unique)
install(TARGETS writy-hyprland LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/writy)
if(BUILD_TESTING)
  add_executable(input-policy-test tests/input_policy_test.cpp)
  target_include_directories(input-policy-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_options(input-policy-test PRIVATE -UNDEBUG)
  add_test(NAME bridge-input-policy COMMAND input-policy-test)
  add_executable(transport-test tests/transport_test.cpp)
  target_include_directories(transport-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_options(transport-test PRIVATE -UNDEBUG)
  add_test(NAME bridge-transport COMMAND transport-test)
endif()
